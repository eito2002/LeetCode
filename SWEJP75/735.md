# [735. Asteroid Collision](https://leetcode.com/problems/asteroid-collision/description/)

## step 1

最初はよくわからなかったが、問題区分がstackであったため、それをヒントにstackで頑張って場合分け。
プラスを詰めて、マイナスが出た時にプラスのものと比較する方針だったが、[-2, -3]のような最初がマイナスの場合、
[-2, -2, 3, -2], [-2, -2, 1, -2]のような場合で何回かつまづき、マイナス側の場合分けを整理していった。(空の場合、前がマイナスの場合はpushすることで解決)
whileとifの順番で混乱し何回か入れ替えた。
```cpp
class Solution {
public:
    vector<int> asteroidCollision(vector<int>& asteroids) {
        stack<int> stk;
        for (int mass : asteroids) {
            if (mass > 0) {
                stk.push(mass);
            } else {
                while (!stk.empty() && abs(stk.top()) < abs(mass) && stk.top() > 0) {
                    stk.pop();
                }
                if (!stk.empty() && abs(stk.top()) == abs(mass) && stk.top() > 0) {
                    stk.pop();
                    continue;
                }
                if (stk.empty() || stk.top() < 0) {
                    stk.push(mass);
                    continue;
                }
            }
        }
        vector<int> vec;
        while (!stk.empty()) {
            vec.push_back(stk.top());
            stk.pop();
        }
        reverse(vec.begin(), vec.end());
        return vec;
    }
};
```

```cpp
class Solution {
public:
    vector<int> asteroidCollision(vector<int>& asteroids) {
        deque<int> deq;
        for (int mass : asteroids) {
            if (mass > 0) {
                deq.push_back(mass);
            } else {
                while (!deq.empty() && abs(deq.back()) < abs(mass) && deq.back() > 0) {
                    deq.pop_back();
                }
                if (!deq.empty() && abs(deq.back()) == abs(mass) && deq.back() > 0) {
                    deq.pop_back();
                } else if (deq.empty() || deq.back() < 0) {
                    deq.push_back(mass);
                }
            }
        }
        vector<int> vec;
        while (!deq.empty()) {
            vec.push_back(deq.front());
            deq.pop_front();
        }
        return vec;
    }
};
```

```cpp
class Solution {
public:
    vector<int> asteroidCollision(vector<int>& asteroids) {
        vector<int> stk;  // 結果用スタックの代わり

        for (int mass : asteroids) {
            bool destroyed = false;

            while (!stk.empty() && mass < 0 && stk.back() > 0) {
                if (stk.back() < -mass) {
                    stk.pop_back();  // 右向きが負けて破壊
                } else if (stk.back() == -mass) {
                    stk.pop_back();  // 両方破壊
                    destroyed = true;
                    break;
                } else {
                    destroyed = true;  // 左向きが破壊
                    break;
                }
            }

            if (!destroyed) {
                stk.push_back(mass);  // 衝突せず生き残った
            }
        }

        return stk;
    }
};
```

爆発しかたどうかをフラグを用いて管理し、していない場合は追加。最初からvectorを使うことでそのままreturnできる。
